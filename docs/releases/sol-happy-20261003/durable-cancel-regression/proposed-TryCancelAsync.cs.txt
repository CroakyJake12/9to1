    public async Task<bool> TryCancelAsync(Guid taskId, Guid ownerUserId, DateTimeOffset now, CancellationToken cancellationToken)
    {
        string? expectedPayload;
        await using (var readConnection = await factory.OpenAsync(cancellationToken).ConfigureAwait(false))
        {
            await using var read = readConnection.CreateCommand();
            read.CommandText = "SELECT payload_json FROM external_agent_tasks WHERE id=$id;";
            read.Parameters.AddWithValue("$id", taskId.ToString());
            expectedPayload = await read.ExecuteScalarAsync(cancellationToken).ConfigureAwait(false) as string;
        }
        if (expectedPayload is null) return false;
        var existing = UnifiedPersistenceJson.Read<ExternalAgentTask>(expectedPayload);
        if (existing.OwnerUserId != ownerUserId || existing.Status is HavenTaskStatus.Completed or HavenTaskStatus.Cancelled) return false;
        await using var connection = await factory.OpenAsync(cancellationToken).ConfigureAwait(false);
        await using var command = connection.CreateCommand();
        command.CommandText = """
            UPDATE external_agent_tasks SET status=$cancelled,payload_json=$payload,updated_at=$updated
            WHERE id=$id AND owner_user_id=$owner AND payload_json=$expected
              AND status<>$completed AND status<>$cancelled;
            """;
        command.Parameters.AddWithValue("$id", taskId.ToString());
        command.Parameters.AddWithValue("$owner", ownerUserId.ToString());
        command.Parameters.AddWithValue("$expected", expectedPayload);
        command.Parameters.AddWithValue("$completed", (int)HavenTaskStatus.Completed);
        command.Parameters.AddWithValue("$cancelled", (int)HavenTaskStatus.Cancelled);
        command.Parameters.AddWithValue("$payload", UnifiedPersistenceJson.Write(existing with { Status = HavenTaskStatus.Cancelled, UpdatedAt = now }));
        command.Parameters.AddWithValue("$updated", now.ToString("O"));
        return await command.ExecuteNonQueryAsync(cancellationToken).ConfigureAwait(false) == 1;
    }
