using System.Security.Cryptography;
using System.Text.Json;
using Avalonia;
using Avalonia.Headless;
using Avalonia.Media.Imaging;
using Avalonia.Skia;
using HavenOS.Images;

var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
var cases = new[] { "canonical-open-source", "canonical-edit-and-reopen", "native-crop-transform-export", "metadata-privacy-export", "source-change-refusal", "overwrite-refusal", "invalid-crop-preserves-state", "future-schema-refusal" };
var outcomes = cases.ToDictionary(x => x, x => new Outcome("NOT-RUN", null));
var assertions = 0;
var metadataReadbacks = new List<object>();
void Require(bool condition, string message) { assertions++; if (!condition) throw new Exception(message); }
void Test(string id, Action execute) { try { execute(); outcomes[id] = new("PASS", null); } catch (Exception error) { outcomes[id] = new("FAIL", error.ToString()); throw; } }
void Refuses<T>(Action execute) where T : Exception { try { execute(); } catch (T) { assertions++; return; } throw new Exception("Expected " + typeof(T).Name); }
string? failure = null;
try
{
    AppBuilder.Configure<Application>().UseSkia().UseHeadless(new AvaloniaHeadlessPlatformOptions { UseHeadlessDrawing = false }).SetupWithoutStarting();
    var sourcePath = Path.Combine(output, "original.png"); File.Copy(args[1], sourcePath, overwrite: false);
    using (var file = TagLib.File.Create(sourcePath))
    {
        var tag = (TagLib.Png.PngTag)file.GetTag(TagLib.TagTypes.Png, true);
        tag.Title = "Real source title"; tag.Comment = "Retained non-location comment";
        var xmp = (TagLib.Xmp.XmpTag)file.GetTag(TagLib.TagTypes.XMP, true);
        xmp.Latitude = 51.5; xmp.Longitude = -.12; xmp.Altitude = 24;
        file.Save();
    }
    var sourceBytes = File.ReadAllBytes(sourcePath); var engine = new PictureCropService(); var original = engine.OpenSource(sourcePath);
    Test(cases[0], () =>
    {
        using var metadataSource = (TagLib.Image.File)TagLib.File.Create(sourcePath);
        metadataReadbacks.Add(new { name = "source", metadataSource.ImageTag.Latitude, metadataSource.ImageTag.Longitude, metadataSource.ImageTag.Altitude });
        Require(metadataSource.ImageTag.Latitude == 51.5 && metadataSource.ImageTag.Longitude == -.12 && metadataSource.ImageTag.Altitude == 24, "Actual persisted source GPS exactly matches fixture before owner export");
        Require(original.DocumentId != Guid.Empty && original.FileId is null, "Real local identity, no hosted FileId");
        Require(original.Revision == 0 && original.CanvasWidth == 8 && original.CanvasHeight == 6 && original.Operations.Count == 0, "Canonical original geometry");
        Require(original.SourceRevision == Convert.ToHexString(SHA256.HashData(sourceBytes)), "Actual original bytes SHA256");
    });
    var edited = original.Crop(1, 2, 5, 3).Rotate().Flip(true);
    Test(cases[1], () =>
    {
        var reopened = PictureDocument.Deserialize(edited.Serialize());
        Require(reopened.DocumentId == original.DocumentId && reopened.SourceRevision == original.SourceRevision && reopened.SourcePath == original.SourcePath, "Identity and original revision preserved");
        Require(reopened.Revision == 3 && reopened.CanvasWidth == 3 && reopened.CanvasHeight == 5 && reopened.Operations.Count == 3, "Canonical crop rotate flip graph survives serialization");
        Require(reopened.Operations[0] == new CropOperation(1, 2, 5, 3) && reopened.Operations[1] == new RotateOperation(1) && reopened.Operations[2] == new FlipOperation(true), "Exact typed operations");
        File.WriteAllBytes(Path.Combine(output, "edited.picture.json"), reopened.Serialize());
    });
    Test(cases[2], () =>
    {
        var path = Path.Combine(output, "transformed.png"); engine.ExportPng(edited, path, PictureMetadataExportMode.RemoveAll);
        using var bitmap = new Bitmap(path); Require(bitmap.PixelSize == new PixelSize(3, 5), "Actual decoded transformed PNG geometry");
        var resized = edited.Resize(6, 10); engine.ExportPng(resized, Path.Combine(output, "resized.png"), PictureMetadataExportMode.RemoveAll);
        using var result = new Bitmap(Path.Combine(output, "resized.png")); Require(result.PixelSize == new PixelSize(6, 10), "Actual decoded resized PNG geometry");
        engine.ExportPng(original, Path.Combine(output, "alpha-roundtrip.png"), PictureMetadataExportMode.RemoveAll);
        Require(File.ReadAllBytes(sourcePath).SequenceEqual(sourceBytes), "Native render/export never alters original bytes");
    });
    Test(cases[3], () =>
    {
        foreach (var mode in Enum.GetValues<PictureMetadataExportMode>())
        {
            var path = Path.Combine(output, mode + ".png"); engine.ExportPng(original, path, mode);
            using var file = TagLib.File.Create(path);
            if (mode == PictureMetadataExportMode.RemoveAll) Require(file.GetTag(TagLib.TagTypes.Png, false) is null, "RemoveAll removes PNG tags");
            else
            {
                var tag = (TagLib.Png.PngTag)file.GetTag(TagLib.TagTypes.Png, false);
                Require(tag.Title == "Real source title" && tag.Comment == "Retained non-location comment", "Non-location metadata retained");
                var image = (TagLib.Image.File)file;
                metadataReadbacks.Add(new { name = mode.ToString(), image.ImageTag.Latitude, image.ImageTag.Longitude, image.ImageTag.Altitude });
                if (mode == PictureMetadataExportMode.RemoveLocation) Require(image.ImageTag.Latitude is null && image.ImageTag.Longitude is null && image.ImageTag.Altitude is null, "Location removed");
                else Require(image.ImageTag.Latitude == 51.5 && image.ImageTag.Longitude == -.12 && image.ImageTag.Altitude == 24, "Location preserved");
            }
        }
        Require(File.ReadAllBytes(sourcePath).SequenceEqual(sourceBytes), "Metadata export never changes source");
    });
    Test(cases[4], () =>
    {
        var changed = sourceBytes.ToArray(); changed[^1] ^= 1; File.WriteAllBytes(sourcePath, changed);
        var destination = Path.Combine(output, "must-not-exist.png");
        try { Refuses<InvalidDataException>(() => engine.ExportPng(original, destination)); Require(!File.Exists(destination), "Changed source creates no partial export"); }
        finally { File.WriteAllBytes(sourcePath, sourceBytes); }
    });
    Test(cases[5], () =>
    {
        var destination = Path.Combine(output, "transformed.png"); var prior = File.ReadAllBytes(destination);
        Refuses<IOException>(() => engine.ExportPng(edited, destination)); Refuses<IOException>(() => engine.ExportPng(edited, sourcePath));
        Require(File.ReadAllBytes(destination).SequenceEqual(prior) && File.ReadAllBytes(sourcePath).SequenceEqual(sourceBytes), "Existing output and source preserved");
    });
    Test(cases[6], () => { var before = edited.Serialize(); Refuses<ArgumentOutOfRangeException>(() => edited.Crop(2, 0, 2, 1)); Require(edited.Serialize().SequenceEqual(before), "Invalid crop immutable prior revision"); });
    Test(cases[7], () => { var text = System.Text.Encoding.UTF8.GetString(edited.Serialize()).Replace("\"schemaVersion\":1", "\"schemaVersion\":99", StringComparison.Ordinal); Refuses<InvalidDataException>(() => PictureDocument.Deserialize(System.Text.Encoding.UTF8.GetBytes(text))); });
}
catch (Exception error) { failure = error.ToString(); Environment.ExitCode = 1; }
finally
{
    var report = new { scope = "Actual unchanged portable Picture owner engine + real native Skia Bitmap; NOT browser or full Picture acceptance", outcomes, assertions, metadataReadbacks, failure,
        counts = new { discovered = cases.Length, passed = outcomes.Count(x => x.Value.Status == "PASS"), failed = outcomes.Count(x => x.Value.Status == "FAIL"), notRun = outcomes.Count(x => x.Value.Status == "NOT-RUN") }, exit = Environment.ExitCode };
    File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(report, new JsonSerializerOptions { WriteIndented = true }));
    Console.WriteLine(JsonSerializer.Serialize(report));
}
record Outcome(string Status, string? Error);
